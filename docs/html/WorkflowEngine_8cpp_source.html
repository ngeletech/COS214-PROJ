191void WorkflowEngine::assignWork(WorkflowInstance &instance, WorkElement *element)
192{

198 AssignmentStrategy *rule = instance.getDefinition()->getAssignmentStrategy();
199 AssignmentStrategy *previous = assignment->getStrategy();
200 if (rule != nullptr)

221void WorkflowEngine::activateNext(WorkflowInstance &instance)
222{
223 if (instance.getStateName() != "Running")
224 {
225 return;
226 }
227
228 std::vector<WorkElement *> ready;
229 AvailableItemIterator it(instance.getRoot());
230 for (it.first(); !it.isDone(); it.next())
231 {
232 ready.push_back(it.current());
233 }
234
235 for (size_t i = 1; i < ready.size(); ++i)
236 {
237 WorkElement *current = ready[i];
238 size_t j = i;
239 while (j > 0 && ready[j - 1]->getPriority() < current->getPriority())
240 {
241 ready[j] = ready[j - 1];
242 --j;
243 }
244 ready[j] = current;
245 }
246
247 for (std::vector<WorkElement *>::iterator rit = ready.begin(); rit != ready.end(); ++rit)
248 {
249 WorkElement *element = *rit;
250 element->setStatus(WorkItemStatus::AVAILABLE);
251
252 if (element->requiresAssignee())
253 {
254 assignWork(instance, element);
255
256 if (instance.getStateName() != "Running")
257 {
258 return;
259 }
260 }
261 }
262 instance.getRoot()->execute();
263
264 if (instance.getStateName() == "Running" && instance.getRoot()->isComplete()) {
265 instance.complete();
