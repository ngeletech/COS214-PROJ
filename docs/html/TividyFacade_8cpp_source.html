159void TividyFacade::decideItem(int id, const std::string& elementName, bool approved) {
160 WorkflowInstance& inst = requireInstance(id);
161 if (inst.getStateName() != "Running")
162 throw std::logic_error("Cannot decide an item while " + inst.getStateName());
163 ApprovalTask* task = dynamic_cast<ApprovalTask*>(inst.findElement(elementName));
164 if (task == nullptr)
165 throw std::invalid_argument("Not an approval item: " + elementName);
166
167 if(!isActive(task->getStatus())) {
168 throw std::logic_error("Item is not active: " + elementName);
169 }
170 task->recordDecision(approved);
171 if (approved) {
172 engine->completeWorkItem(id, elementName);
173 } else {
174 Issue issue("Rejected", elementName + " was rejected", 3, elementName);
175 engine->handleIssue(issue, inst);
176 }
177}

196std::vector<std::string> TividyFacade::getAvailableWork(int id) const {
197 std::vector<std::string> names;
198 WorkflowInstance& inst = requireInstance(id);
199
200 if (inst.getRoot() == nullptr) {
201 return names;
202 }
203
204 for (DepthFirstIterator it(inst.getRoot()); !it.isDone(); it.next()) {
205 WorkElement* e = it.current();
206 if (dynamic_cast<WorkGroup*>(e) != nullptr) continue; // skip groups
207 WorkItemStatus s = e->getStatus();
208 if (s == WorkItemStatus::AVAILABLE || s == WorkItemStatus::ASSIGNED ||s == WorkItemStatus::STARTED) {
209 names.push_back(e->getName());
210 }
211 }
212
213 return names;
214}

216std::string TividyFacade::generateReport(int id) const {
217 WorkflowInstance& inst = requireInstance(id);
218 std::ostringstream out;
219
220 out << "=== Workflow instance " << id << " ===\n";
221 out << "State: " << inst.getStateName() << "\n";
222
223 // data
224 out << "\nData:\n";
225 const std::map<std::string, std::string>& values = inst.getData().getData();
226 if (values.empty()) {
227 out << " (none)\n";
228 }
229 for (std::map<std::string, std::string>::const_iterator it = values.begin();
230 it != values.end(); ++it) {
231 out << " " << it->first << " = " << it->second << "\n";
232 }
233
234 // progress: every task in the work tree
235 out << "\nWork items:\n";
236 for (DepthFirstIterator it(inst.getRoot()); !it.isDone(); it.next()) {
237 WorkElement* element = it.current();
238 if (element == inst.getRoot()) continue; // skip the hidden root group
239 out << " " << element->getName() << ": " << statusText(element->getStatus());
240 Participant* who = element->getAssignee();
241 if (who != nullptr) out << " (" << who->getName() << ")";
242 out << "\n";
243 }
244
245 // history: the audit entries for this instance only
246 out << "\nHistory:\n";
247 std::unique_ptr<WorkIterator<const AuditEntry> > entries(audit->createIterator());
248 int count = 0;
249 for (entries->first(); !entries->isDone(); entries->next()) {
250 const AuditEntry* entry = entries->current();
251 if (entry->getInstanceId() != id) continue;
252 out << " " << entry->getEventType();
253 if (!entry->getDetails().empty()) out << " - " << entry->getDetails();
254 out << "\n";
255 ++count;
256 }
257 if (count == 0) out << " (no events recorded)\n";
258
259 return out.str();
260}
