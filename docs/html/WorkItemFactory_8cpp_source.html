16 /**
17 * @brief Removes the leading and trailing spaces
18 * and tabs from a string*/
19 std::string trim(const std::string& text) {
20 const char* whitespace = " \t";
21 std::string::size_type first = text.find_first_not_of(whitespace);
22 if (first == std::string::npos) {
23 return "";
24 }
25 std::string::size_type last = text.find_last_not_of(whitespace);
26 return text.substr(first, last - first + 1);

80/**
81 * @brief The specs config is a single string for a script task it
82 * is assumed to look like "key,value"; only the first ","
83 * splits it.
84 * @details Uses new ScriptTask(spec.name, key, value)
85 */
86WorkElement* ScriptTaskFactory::makeElement(const WorkItemSpec& spec){
87 std::string::size_type comma = spec.config.find(',');
88 if(comma == std::string::npos){
89 throw std::invalid_argument("ScriptTask config must look like \"key,value\"");
90 }
91 std::string key = trim(spec.config.substr(0, comma));
92 std::string value = trim(spec.config.substr(comma + 1));
93
94 if(key.empty()) {
95 throw std::invalid_argument("ScriptTask config needs a non-empty key");
96 }
97 return new ScriptTask(spec.name, key, value);
98}
