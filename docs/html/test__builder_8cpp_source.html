13 TEST_CASE("WorkflowDesigner - Assembly via Director") {

26 //Director contructs the product

29 REQUIRE(def != nullptr);

32 CHECK(def->getWorkItemSpecs()[0].name == "ReviewApplication");
33 CHECK(def->getWorkItemSpecs()[1].name == "SendConfirmationEmail");
34 CHECK(def->getAssignmentStrategy() == strategy);

37 //safely deletes strategy and handler internally
