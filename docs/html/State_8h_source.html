2 * @file State.h
3 * @brief State pattern for the workflow instance lifecycle.
4 *
5 * Pattern roles:
6 * - Context: WorkflowInstance
7 * - State: InstanceState
8 * - ConcreteState: CreatedState, RunningState, SuspendedState,
9 * CompletedState, CancelledState, FailedState
10 *
11 * WorkflowInstance also takes part in the Observer pattern (as Subject) and
12 * the Memento pattern (as Originator).
13 */

33/**
34 * @class InstanceState
35 * @brief Abstract State. Declares every lifecycle action.
36 *
37 * Every action throws std::logic_error by default. A concrete state
38 * overrides only the actions that are legal while in that state.
39 */
40class InstanceState {
41protected:
42 /** @brief The context this state controls (not owned). */
43 WorkflowInstance* instance;
44
45public:
46 /**
47 * @brief Binds the state to its context.
48 * @param i The WorkflowInstance this state controls.
49 */
50 InstanceState(WorkflowInstance& i);
51
52 /** @brief Virtual destructor so states can be deleted through an InstanceState pointer. */
53 virtual ~InstanceState() {}
54
55 /**
56 * @brief Moves Created -> Running.
57 * @throws std::logic_error if not allowed in the current state.
58 */
59 virtual void start();
60
61 /**
62 * @brief Moves Running -> Suspended.
63 * @throws std::logic_error if not allowed in the current state.
64 */
65 virtual void suspend();
66
67 /**
68 * @brief Moves Suspended -> Running.
69 * @throws std::logic_error if not allowed in the current state.
70 */
71 virtual void resume();
72
73 /**
74 * @brief Moves Running -> Completed.
75 * @throws std::logic_error if not allowed in the current state.
76 */
77 virtual void complete();
78
79 /**
80 * @brief Moves to Cancelled and cancels outstanding work.
81 * @throws std::logic_error if not allowed in the current state.
82 */
83 virtual void cancel();
84
85 /**
86 * @brief Records a failure and moves to Failed.
87 * @param issue The problem that caused the failure.
88 * @throws std::logic_error if not allowed in the current state.
89 */
90 virtual void fail(const Issue& issue);
91
92 /**
93 * @brief Checks whether cancel() is allowed in this state.
94 * @return false by default; true in Created, Running and Suspended.
95 */
96 virtual bool canCancel() const;
97
98 /**
99 * @brief Gets the state name, used by the Memento to save and restore it.
100 * @return The state name, e.g. "Created" or "Running".
101 */
102 virtual std::string getName() const = 0;
103
104 /**
105 * @brief Factory used when restoring from a snapshot.
106 * @param name Value previously returned by getName().
107 * @param inst Context the new state will belong to.
108 * @return A newly allocated state; the caller takes ownership.
109 * @throws std::invalid_argument if @p name is unknown.
110 */
111 static InstanceState* create(const std::string& name, WorkflowInstance& inst);
112};

254/**
255 * @class WorkflowInstance
