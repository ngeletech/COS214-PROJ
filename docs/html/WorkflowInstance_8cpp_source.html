168void WorkflowInstance::restore(const InstanceSnapshot& snap) {
169 data = snap.data;
170
171 if (root != nullptr) {
172 for (DepthFirstIterator it(root); !it.isDone(); it.next()) {
173 WorkElement* element = it.current();
174
175 if(holdsWork(element)) element->getAssignee()->decrementWorkload();
176 if (engine != nullptr) {
177 Participant* who = nullptr;
178 std::map<std::string, int>::const_iterator a = snap.assignments.find(element->getName());
179 if (a != snap.assignments.end()) who = engine->findParticipant(a->second);
180 element->assignTo(who);
181 }
182
183 // set the status last, because assignTo may change it
184 std::map<std::string, WorkItemStatus>::const_iterator s = snap.elementStatuses.find(element->getName());
185 if (s != snap.elementStatuses.end()) {
186 element->setStatus(s->second);
187 }
188
189 if(holdsWork(element)) element->getAssignee()->incrementWorkload();
190 }
191 }
192
193 setState(InstanceState::create(snap.stateName, *this));
194}
