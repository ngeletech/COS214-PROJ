21/**
22 * @class AssignmentStrategy
23 * @brief Abstract strategy interface for selecting a participant to perform a work item.
24 *
25 * Concrete strategies implement different participant-selection algorithms.
26 * The selected participant is returned to the caller; this interface does
27 * not perform the actual work assignment.
28 */
29class AssignmentStrategy {
30 public:
31 /**
32 * @brief Selects a participant from the available candidates.
33 *
34 * @param element Work element for which a participant is being selected.
35 * @param data Data associated with the work item.
36 * @param candidates List of participants considered for assignment.
37 * @return Pointer to the selected participant, or nullptr if no
38 * suitable candidate is found.
39 */
40 virtual Participant* select(WorkElement& element, const WorkData& data, std::vector<Participant*> candidates) = 0;
41
42 /** @brief Virtual destructor for safe destruction through the base class. */
43 virtual ~AssignmentStrategy() {}
44};

142/**
143 * @class AssignmentService
144 * @brief Context class that uses an assignment strategy to select participants.
145 *
146 * Maintains a pointer to the currently selected AssignmentStrategy.
147 * The strategy can be changed at runtime, allowing different participant-
148 * selection algorithms to be used without changing the service itself.
149 */
150class AssignmentService {
151 private:
152 AssignmentStrategy* strategy; ///< Strategy currently used for participant selection.
153
154 public:
155 /**
156 * @brief Constructs the assignment service with an initial strategy.
157 * @param strategy Strategy used to select participants.
158 */
159 AssignmentService(AssignmentStrategy* strategy) : strategy(strategy) {}
160
161 /**
162 * @brief Replaces the currently configured assignment strategy.
163 * @param newStrategy Strategy to use for subsequent selections.
164 */
165 void setStrategy(AssignmentStrategy* newStrategy) { strategy = newStrategy; }
166
167 /**
168 * @brief Returns the currently configured assignment strategy.
169 * @return Pointer to the current strategy.
170 */
171 AssignmentStrategy* getStrategy() const { return strategy; }
172
173 /**
174 * @brief Selects a participant by delegating to the current strategy.
175 *
176 * @param element Work element for which a participant is being selected.
177 * @param data Data associated with the work item.
178 * @param candidates Participants available for selection.
179 * @return Participant selected by the configured strategy.
180 *
181 * @note This method does not assign the work item to the returned
182 * participant; it only delegates participant selection.
183 */
184 Participant* assignWorkItem(WorkElement& element, const WorkData& data, std::vector<Participant*> candidates);
185};
